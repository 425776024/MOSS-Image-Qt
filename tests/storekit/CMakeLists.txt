cmake_minimum_required(VERSION 3.24)
project(StoreKitChecks LANGUAGES Swift)
set(CMAKE_OSX_DEPLOYMENT_TARGET 26.0)
set(CMAKE_Swift_LANGUAGE_VERSION 5)
find_package(XCTest REQUIRED)
add_executable(StoreKitHost MACOSX_BUNDLE Host.swift)
set_target_properties(StoreKitHost PROPERTIES MACOSX_BUNDLE_GUI_IDENTIFIER ai.ffclip.studio.storekit-host XCODE_GENERATE_SCHEME TRUE XCODE_ATTRIBUTE_CODE_SIGNING_ALLOWED YES XCODE_ATTRIBUTE_CODE_SIGN_IDENTITY "-")
target_link_libraries(StoreKitHost PRIVATE "-framework AppKit")
xctest_add_bundle(StoreKitChecks StoreKitHost StorePurchasesTests.swift ../../src/apple/StorePurchases.swift)
set_target_properties(StoreKitChecks PROPERTIES XCODE_GENERATE_SCHEME TRUE XCODE_ATTRIBUTE_PRODUCT_BUNDLE_IDENTIFIER ai.ffclip.studio.storekit-tests XCODE_ATTRIBUTE_CODE_SIGNING_ALLOWED YES XCODE_ATTRIBUTE_CODE_SIGN_IDENTITY "-")
target_link_libraries(StoreKitChecks PRIVATE "-framework StoreKitTest" "-framework StoreKit" "-framework AppKit" "-framework Security")
target_compile_options(StoreKitChecks PRIVATE "-F${XCTest_LIBRARY}/..")
set_source_files_properties(Products.storekit PROPERTIES MACOSX_PACKAGE_LOCATION Resources)
target_sources(StoreKitChecks PRIVATE Products.storekit)

set_target_properties(StoreKitChecks PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
